from flask import Flask, render_template, request, redirect, url_for from database import ( get_db, create_database, add_sample_listings, add_listing ) app = Flask(__name__) # ========================= # تجهيز قاعدة البيانات # ========================= create_database() add_sample_listings() # ========================= # الصفحة الرئيسية # ========================= @app.route("/") def home(): connection = get_db() listings = connection.execute(""" SELECT * FROM listings ORDER BY id DESC """).fetchall() connection.close() return render_template( "index.html", listings=listings, search_query="" ) # ========================= # البحث # ========================= @app.route("/search") def search(): query = request.args.get("q", "").strip() connection = get_db() if query: listings = connection.execute(""" SELECT * FROM listings WHERE title LIKE ? OR category LIKE ? OR city LIKE ? OR description LIKE ? ORDER BY id DESC """, ( f"%{query}%", f"%{query}%", f"%{query}%", f"%{query}%" )).fetchall() else: listings = connection.execute(""" SELECT * FROM listings ORDER BY id DESC """).fetchall() connection.close() return render_template( "index.html", listings=listings, search_query=query ) # ========================= # صفحة إضافة إعلان # ========================= @app.route("/sell") def sell(): return render_template("sell.html") # ========================= # حفظ الإعلان الجديد # ========================= @app.route("/sell", methods=["POST"]) def create_listing(): title = request.form.get("title", "").strip() category = request.form.get("category", "").strip() price = request.form.get("price", "").strip() city = request.form.get("city", "").strip() description = request.form.get("description", "").strip() # التأكد من الحقول الأساسية if not title or not category or not price or not city: return "فضلاً عبّ جميع الحقول المطلوبة." # تحويل السعر إلى رقم try: price = float(price) except ValueError: return "السعر يجب أن يكون رقمًا." # إضافة الإعلان إلى قاعدة البيانات add_listing( title, category, price, city, description ) # العودة للرئيسية return redirect(url_for("home")) # ========================= # صفحة تفاصيل الإعلان # ========================= @app.route("/listing/") def listing_details(listing_id): connection = get_db() listing = connection.execute(""" SELECT * FROM listings WHERE id = ? """, (listing_id,)).fetchone() connection.close() # إذا الإعلان غير موجود if listing is None: return "الإعلان غير موجود", 404 # فتح صفحة الإعلان return render_template( "listing.html", listing=listing ) # ========================= # تشغيل الموقع # ========================= if __name__ == "__main__": app.run(debug=True)
